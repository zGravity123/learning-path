// MAIN.CPP

#include <iostream>
#include <windows.h>
#include "Validacao.h"

using namespace std;

int main()
{
    SetConsoleCP(1252);
    SetConsoleOutputCP(1252);
	
    double nota;
	
    cout << "Introduza uma nota: ";
    cin >> nota;
	
    try
    {
        validarNota(nota);
		
        cout << "Nota valida: " << nota << endl;
		
        if (nota >= 10)
            cout << "Aluno aprovado." << endl;
        else
            cout << "Aluno Reprovado." << endl;
    }
    catch (const char* mensagem)
    {
        cout << "Erro: " << mensagem << endl;
    }

    return 0;
}

// VALIDACAO.CPP

#include "Validacao.h"

void validarNota(double nota)
{
	if (nota < 0 || nota > 20)
	{
		throw "A nota tem de estar entre 0 e 20.";
	}
}
